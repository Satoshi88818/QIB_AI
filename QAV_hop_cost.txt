//! Pure hop-cost functions. These are the *real* functions used by the router
//! and called directly by the property tests (v17's proofs re-implemented the
//! logic inline, so they proved a copy rather than the code).
use crate::config::{RuntimeConfig, HOP_COST_FLOOR};

#[derive(Clone, Copy, Debug, PartialEq)]
pub struct HopInputs {
    pub mean: f64,
    pub std: f64,
    pub quarantined: bool,
}

impl HopInputs {
    /// Inputs for a link the twin has never seen: maximum uncertainty.
    pub const UNKNOWN: HopInputs = HopInputs { mean: 0.0, std: 1.0, quarantined: false };
}

#[derive(Clone, Copy, Debug, PartialEq)]
pub struct CostParams {
    pub tax_threshold: f64,
    pub reject_threshold: f64,
    pub tax_multiplier: f64,
}

impl CostParams {
    pub fn standard(c: &RuntimeConfig) -> Self {
        Self {
            tax_threshold: c.safety_tax_threshold,
            reject_threshold: c.path_reject_threshold,
            tax_multiplier: c.safety_tax_multiplier,
        }
    }
    pub fn conservative(c: &RuntimeConfig) -> Self {
        Self {
            tax_threshold: c.safety_tax_threshold,
            reject_threshold: c.conservative_reject_threshold,
            tax_multiplier: c.conservative_tax_multiplier,
        }
    }
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum GateClass {
    Clear,
    Taxed,
    Rejected,
}

/// Classify a link's uncertainty. Anything non-finite is rejected (fail closed).
pub fn gate_class(i: &HopInputs, p: &CostParams) -> GateClass {
    if !(i.std.is_finite() && i.std >= 0.0 && i.mean.is_finite()) || i.std > p.reject_threshold {
        GateClass::Rejected
    } else if i.std > p.tax_threshold {
        GateClass::Taxed
    } else {
        GateClass::Clear
    }
}

/// AI-mode hop cost. `f64::INFINITY` means "never route over this link".
/// Quarantine beats everything, regardless of std.
pub fn hop_cost(i: &HopInputs, p: &CostParams) -> f64 {
    if i.quarantined {
        return f64::INFINITY;
    }
    match gate_class(i, p) {
        GateClass::Rejected => f64::INFINITY,
        GateClass::Taxed => ((1.0 - i.mean.clamp(0.0, 1.0)) + HOP_COST_FLOOR) * p.tax_multiplier,
        GateClass::Clear => (1.0 - i.mean.clamp(0.0, 1.0)) + HOP_COST_FLOOR,
    }
}

/// Safe-mode hop cost: static topology weight, but quarantine still applies.
pub fn safe_hop_cost(static_weight: f64, quarantined: bool) -> f64 {
    if quarantined || !(static_weight.is_finite() && static_weight > 0.0) {
        f64::INFINITY
    } else {
        static_weight
    }
}

//! IMP-P: probabilistic twin (time-decayed Gaussian per directed link).
//!
//! Fixes vs v17:
//! * Global uncertainty stats are maintained as `sum == Σ per-link contribution`
//!   under one lock; v17's incremental sum subtracted a phantom 1.0 for every new
//!   link and collapsed to ~0, blinding the spike detector and health score.
//! * `uncertainty` no longer shrinks as 1/√n: it is the *predictive* std of a
//!   decay-weighted Gaussian, so a link that turns flaky stays visibly uncertain.
//! * Non-finite / out-of-range observations are rejected, not clamped (clamp
//!   propagates NaN).
//! * Link keys are `(src, dst)` pairs, not `"src→dst"` strings.
//! * Getters never create entries.
use crate::config::{MIN_SAMPLES_FOR_CONFIDENCE};
use crate::errors::{Result, VerifierError};
use crate::hop_cost::HopInputs;
use std::collections::HashMap;
use std::fmt;
use std::sync::RwLock;

#[derive(Clone, Debug, PartialEq, Eq, Hash, PartialOrd, Ord)]
pub struct LinkId {
    pub src: String,
    pub dst: String,
}

impl LinkId {
    pub fn new(src: &str, dst: &str) -> Self {
        Self { src: src.to_owned(), dst: dst.to_owned() }
    }
}

impl fmt::Display for LinkId {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        write!(f, "{}->{}", self.src, self.dst)
    }
}

#[derive(Clone, Debug)]
struct LinkState {
    mean: f64,
    /// Decay-weighted sum of squared deviations.
    s: f64,
    /// Σ weights, Σ weights² (for effective sample size).
    w: f64,
    w2: f64,
    n: u64,
    quarantined: bool,
    quarantine_reason: Option<String>,
    brier_ew: f64,
    brier_n: u64,
    /// This link's current contribution to the global std sum.
    contrib: f64,
}

impl LinkState {
    fn fresh() -> Self {
        Self {
            mean: 0.0,
            s: 0.0,
            w: 0.0,
            w2: 0.0,
            n: 0,
            quarantined: false,
            quarantine_reason: None,
            brier_ew: 0.0,
            brier_n: 0,
            contrib: 1.0, // matches std() of a fresh link, and is added to the sum on insert
        }
    }

    /// Predictive std. 1.0 (maximum uncertainty) until enough samples exist.
    fn std(&self) -> f64 {
        if self.n < MIN_SAMPLES_FOR_CONFIDENCE || !(self.w > 0.0 && self.w2 > 0.0) {
            return 1.0;
        }
        let var = (self.s / self.w).max(0.0);
        let n_eff = (self.w * self.w) / self.w2;
        let sd = (var * (1.0 + 1.0 / n_eff)).sqrt();
        if sd.is_finite() {
            sd.min(1.0)
        } else {
            1.0
        }
    }

    fn update(&mut self, x: f64, decay_alpha: f64) {
        let d = 1.0 - decay_alpha;
        self.w = self.w * d + 1.0;
        self.w2 = self.w2 * d * d + 1.0;
        self.s *= d;
        let delta = x - self.mean;
        self.mean += delta / self.w;
        self.s += delta * (x - self.mean);
        if self.s < 0.0 {
            self.s = 0.0; // fp noise
        }
        self.n += 1;
    }

    fn brier(&self, window: u64) -> Option<f64> {
        (self.brier_n >= window).then_some(self.brier_ew)
    }
}

#[derive(Clone, Debug, PartialEq)]
pub struct LinkSnapshot {
    pub mean: f64,
    pub std: f64,
    pub samples: u64,
    pub quarantined: bool,
    pub brier: Option<f64>,
}

#[derive(Clone, Debug, PartialEq)]
pub struct ObserveResult {
    pub std_before: f64,
    pub snapshot: LinkSnapshot,
}

#[derive(Default)]
struct Inner {
    links: HashMap<LinkId, LinkState>,
    std_sum: f64,
    ops: u64,
}

impl Inner {
    fn entry(&mut self, link: &LinkId) -> &mut LinkState {
        if !self.links.contains_key(link) {
            let st = LinkState::fresh();
            self.std_sum += st.contrib;
            self.links.insert(link.clone(), st);
        }
        // Just ensured present; the fallback arm is unreachable but avoids unwrap.
        self.links.entry(link.clone()).or_insert_with(LinkState::fresh)
    }

    fn recompute_sum(&mut self) {
        self.std_sum = self.links.values().map(|s| s.contrib).sum();
    }
}

fn snap(s: &LinkState, brier_window: u64) -> LinkSnapshot {
    LinkSnapshot {
        mean: s.mean,
        std: s.std(),
        samples: s.n,
        quarantined: s.quarantined,
        brier: s.brier(brier_window),
    }
}

pub struct TwinStore {
    inner: RwLock<Inner>,
}

impl Default for TwinStore {
    fn default() -> Self {
        Self::new()
    }
}

const RECOMPUTE_EVERY: u64 = 1024;

impl TwinStore {
    pub fn new() -> Self {
        Self { inner: RwLock::new(Inner::default()) }
    }

    fn w(&self) -> Result<std::sync::RwLockWriteGuard<'_, Inner>> {
        self.inner.write().map_err(|_| VerifierError::LockPoisoned)
    }
    fn r(&self) -> Result<std::sync::RwLockReadGuard<'_, Inner>> {
        self.inner.read().map_err(|_| VerifierError::LockPoisoned)
    }

    pub fn observe(&self, link: &LinkId, x: f64, decay_alpha: f64, brier_window: u64) -> Result<ObserveResult> {
        if !(x.is_finite() && (0.0..=1.0).contains(&x)) {
            return Err(VerifierError::InvalidObservation(x));
        }
        let mut g = self.w()?;
        let std_before = g.entry(link).std();
        let (new_std, snapshot) = {
            let st = g.entry(link);
            st.update(x, decay_alpha);
            let ns = st.std();
            (ns, snap(st, brier_window))
        };
        let old_contrib = {
            let st = g.entry(link);
            let o = st.contrib;
            st.contrib = new_std;
            o
        };
        g.std_sum += new_std - old_contrib;
        g.ops += 1;
        if g.ops % RECOMPUTE_EVERY == 0 {
            g.recompute_sum(); // bound floating-point drift
        }
        Ok(ObserveResult { std_before, snapshot })
    }

    /// Record a prediction-vs-reality comparison. Returns `(delta, predicted_mean)`.
    pub fn compare(&self, link: &LinkId, actual: f64, brier_window: u64) -> Result<(f64, f64)> {
        if !(actual.is_finite() && (0.0..=1.0).contains(&actual)) {
            return Err(VerifierError::InvalidObservation(actual));
        }
        let mut g = self.w()?;
        let st = g
            .links
            .get_mut(link)
            .filter(|s| s.n > 0)
            .ok_or_else(|| VerifierError::UnknownLink(link.to_string()))?;
        let predicted = st.mean;
        let err2 = (predicted - actual).powi(2);
        st.brier_n += 1;
        let a = (1.0 / brier_window.max(1) as f64).max(1.0 / st.brier_n as f64);
        st.brier_ew += a * (err2 - st.brier_ew);
        Ok(((predicted - actual).abs(), predicted))
    }

    /// Read-only; never inserts.
    pub fn snapshot(&self, link: &LinkId, brier_window: u64) -> Result<Option<LinkSnapshot>> {
        Ok(self.r()?.links.get(link).map(|s| snap(s, brier_window)))
    }

    pub fn predicted_mean(&self, link: &LinkId) -> Result<Option<f64>> {
        Ok(self.r()?.links.get(link).filter(|s| s.n > 0).map(|s| s.mean))
    }

    pub fn is_quarantined(&self, link: &LinkId) -> Result<bool> {
        Ok(self.r()?.links.get(link).map_or(false, |s| s.quarantined))
    }

    /// Set/clear quarantine. Returns true if the state changed. Creates the link
    /// (at maximum uncertainty) if unknown so the quarantine cannot be lost.
    pub fn set_quarantined(&self, link: &LinkId, value: bool, reason: &str) -> Result<bool> {
        let mut g = self.w()?;
        let st = g.entry(link);
        if st.quarantined == value {
            return Ok(false);
        }
        st.quarantined = value;
        st.quarantine_reason = if value { Some(reason.to_owned()) } else { None };
        Ok(true)
    }

    pub fn quarantine_reason(&self, link: &LinkId) -> Result<Option<String>> {
        Ok(self.r()?.links.get(link).and_then(|s| s.quarantine_reason.clone()))
    }

    /// Hop-cost inputs for many links under ONE read lock (consistent snapshot).
    pub fn hop_inputs_bulk<'a, I: IntoIterator<Item = &'a LinkId>>(&self, links: I) -> Result<Vec<HopInputs>> {
        let g = self.r()?;
        Ok(links
            .into_iter()
            .map(|l| match g.links.get(l) {
                Some(s) => HopInputs { mean: s.mean, std: s.std(), quarantined: s.quarantined },
                None => HopInputs::UNKNOWN,
            })
            .collect())
    }

    /// O(1) mean std across tracked links. Unknown-to-twin ⇒ 1.0.
    pub fn global_avg_std(&self) -> Result<f64> {
        let g = self.r()?;
        if g.links.is_empty() {
            return Ok(1.0);
        }
        Ok((g.std_sum / g.links.len() as f64).clamp(0.0, 1.0))
    }

    pub fn len(&self) -> Result<usize> {
        Ok(self.r()?.links.len())
    }

    pub fn is_empty(&self) -> Result<bool> {
        Ok(self.len()? == 0)
    }

    pub fn quarantine_fraction(&self) -> Result<f64> {
        let g = self.r()?;
        if g.links.is_empty() {
            return Ok(0.0);
        }
        Ok(g.links.values().filter(|s| s.quarantined).count() as f64 / g.links.len() as f64)
    }

    pub fn quarantined_links(&self) -> Result<Vec<LinkId>> {
        let mut v: Vec<LinkId> = self.r()?.links.iter().filter(|(_, s)| s.quarantined).map(|(k, _)| k.clone()).collect();
        v.sort();
        Ok(v)
    }

    pub fn all_links(&self) -> Result<Vec<LinkId>> {
        let mut v: Vec<LinkId> = self.r()?.links.keys().cloned().collect();
        v.sort();
        Ok(v)
    }

    /// Worst-N links by std descending (ties broken by id for determinism).
    pub fn top_n_worst(&self, n: usize) -> Result<Vec<(LinkId, f64)>> {
        let g = self.r()?;
        let mut v: Vec<(LinkId, f64)> = g.links.iter().map(|(k, s)| (k.clone(), s.std())).collect();
        v.sort_by(|a, b| b.1.total_cmp(&a.1).then_with(|| a.0.cmp(&b.0)));
        v.truncate(n);
        Ok(v)
    }

    pub fn any_calibration_drift(&self, threshold: f64, window: u64) -> Result<bool> {
        Ok(self.r()?.links.values().any(|s| s.brier(window).map_or(false, |b| b > threshold)))
    }

    /// Test/diagnostic: does the incremental sum equal the exact sum?
    pub fn std_sum_drift(&self) -> Result<f64> {
        let g = self.r()?;
        let exact: f64 = g.links.values().map(|s| s.contrib).sum();
        Ok((exact - g.std_sum).abs())
    }
}
