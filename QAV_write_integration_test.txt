#![allow(dead_code)]
use ed25519_dalek::SigningKey;
use qib_verifier::audit::MemoryAuditSink;
use qib_verifier::clock::ManualClock;
use qib_verifier::config::RuntimeConfig;
use qib_verifier::graph::Graph;
use qib_verifier::sentinel::SignedProbe;
use qib_verifier::verifier::{ControlPlane, DataPlane, Verifier};
use std::sync::atomic::{AtomicU64, Ordering};
use std::sync::Arc;

pub const BASE_MS: u64 = 1_793_750_400_000; // 2026-11-04T00:00:00Z

pub struct Fx {
    pub dp: DataPlane,
    pub cp: ControlPlane,
    pub audit: Arc<MemoryAuditSink>,
    pub clock: Arc<ManualClock>,
    pub sk: SigningKey,
    nonce: AtomicU64,
}

pub fn fx() -> Fx {
    fx_with(RuntimeConfig::default())
}

pub fn fx_with(cfg: RuntimeConfig) -> Fx {
    let audit = Arc::new(MemoryAuditSink::new());
    let clock = Arc::new(ManualClock::new(BASE_MS));
    let sk = SigningKey::from_bytes(&[42u8; 32]);
    let (dp, cp) = Verifier::build(cfg, sk.verifying_key(), audit.clone(), clock.clone()).unwrap();
    Fx { dp, cp, audit, clock, sk, nonce: AtomicU64::new(1) }
}

impl Fx {
    pub fn probe(&self, src: &str, dst: &str, fidelity: f64) -> SignedProbe {
        let n = self.nonce.fetch_add(1, Ordering::SeqCst);
        SignedProbe::sign(&self.sk, src, dst, fidelity, n, self.clock.unix_ms_now())
    }
    pub fn warm(&self, src: &str, dst: &str, fidelity: f64) {
        for _ in 0..12 {
            self.dp.observe(src, dst, fidelity).unwrap();
        }
    }
    /// Alternate between hi/lo to create a genuinely uncertain link.
    pub fn flaky(&self, src: &str, dst: &str, hi: f64, lo: f64) {
        for i in 0..40 {
            self.dp.observe(src, dst, if i % 2 == 0 { hi } else { lo }).unwrap();
        }
    }
}

pub trait ClockExt {
    fn unix_ms_now(&self) -> u64;
}
impl ClockExt for ManualClock {
    fn unix_ms_now(&self) -> u64 {
        use qib_verifier::clock::Clock;
        self.unix_ms()
    }
}

pub fn diamond() -> Graph {
    Graph::new(vec![
        ("A".into(), "B".into(), 1.0),
        ("B".into(), "D".into(), 1.0),
        ("A".into(), "C".into(), 1.0),
        ("C".into(), "D".into(), 1.0),
        ("A".into(), "D".into(), 5.0),
    ])
    .unwrap()
}

pub fn line() -> Graph {
    Graph::new(vec![("A".into(), "B".into(), 1.0), ("B".into(), "D".into(), 1.0)]).unwrap()
}

/// Warm all diamond links: via-B path better than via-C.
pub fn warm_diamond(f: &Fx) {
    f.warm("A", "B", 0.99);
    f.warm("B", "D", 0.99);
    f.warm("A", "C", 0.92);
    f.warm("C", "D", 0.92);
    f.warm("A", "D", 0.80);
}

pub fn uses_hop(paths: &[qib_verifier::RoutedPath], a: &str, b: &str) -> bool {
    paths.iter().any(|p| p.nodes.windows(2).any(|w| w[0] == a && w[1] == b))
}

    let rejects = f.audit.events().iter().filter(|e| matches!(e, AuditEvent::ProbeRejected { .. })).count();

#[test]
fn drift_escalation_ladder_reaches_diverse_conservative_routing_and_chaos_clears() {
    let f = fx();
    warm_diamond(&f);
    let mut mesh_requested = false;
    for _cycle in 0..3 {
        for _ in 0..25 {
            f.dp.compare_and_audit("A", "B", 0.10).unwrap(); // bad
        }
        assert_eq!(f.dp.learning_rate().unwrap(), LEARNING_RATE_THROTTLED, "overconfidence throttles learning");
        for _ in 0..120 {
            f.dp.compare_and_audit("A", "B", 0.99).unwrap(); // good: ewma falls, hysteresis re-arms
        }
        if f.dp.take_mesh_probe_request().unwrap().is_some() {
            mesh_requested = true;
        }
    }
    let v = f.dp.safety_view().unwrap();
    assert!(v.ai_enabled && v.stage == 3 && v.conservative, "got {v:?}");
    assert!(mesh_requested, "stage 2 must request a full-mesh probe from Rust");
    assert_eq!(v.learning_rate, LEARNING_RATE_NORMAL, "throttle is reversible");

    let r = f.dp.route("A", "D", 2, &diamond()).unwrap();
    assert_eq!(r.mode, RouteMode::AiConservative);
    assert_eq!(r.paths.len(), 2, "stage>=3 yields two edge-disjoint paths");
    let (p0, p1) = (&r.paths[0].nodes, &r.paths[1].nodes);
    assert_ne!(p0, p1);
    assert!(!(uses_hop(&r.paths[..1], "B", "D") && uses_hop(&r.paths[1..], "B", "D")));
    // 4th excursion trips the breaker
    for _ in 0..25 {
        f.dp.compare_and_audit("A", "B", 0.10).unwrap();
    }
    assert!(!f.dp.safety_view().unwrap().ai_enabled);
}

fn health_is_read_only_and_reflects_reality() {

    let h1 = f.dp.health().unwrap();

    f.clock.advance_secs(60);
    f.flaky("A", "B", 0.99, 0.30);
    f.flaky("B", "D", 0.99, 0.30);
    f.dp.tick().unwrap();
    let h3 = f.dp.health().unwrap();

    assert!(h3.mean_uncertainty > h1.mean_uncertainty);
    assert!(h3.ai_trust_score < h1.ai_trust_score);
    assert_eq!(h3.top_n_worst_links.len(), 5);
    assert!(h3.top_n_worst_links.windows(2).all(|w| w[0].1 >= w[1].1));
    assert!(f.dp.safe_routing_coverage_pct(&g).unwrap() < 100.0);
}

//! Whole-system randomized tests through the public API only.
mod common;
use common::*;
use proptest::prelude::*;
use qib_verifier::graph::Graph;

const NODES: [&str; 5] = ["n0", "n1", "n2", "n3", "n4"];

fn complete_graph() -> Graph {
    let mut e = Vec::new();
    for (i, a) in NODES.iter().enumerate() {
        for (j, b) in NODES.iter().enumerate() {
            if i != j {
                e.push((a.to_string(), b.to_string(), 1.0 + ((i + 2 * j) % 4) as f64));
            }
        }
    }
    Graph::new(e).unwrap()
}

#[derive(Debug, Clone)]
enum Op {
    Observe(usize, usize, f64),
    Probe(usize, usize, f64),
    Compare(usize, usize, f64),
    Route(usize, usize, usize),
    Tick(u64),
    Conservative(bool),
    Reset,
}

fn arb_op() -> impl Strategy<Value = Op> {
    let n = NODES.len();
    prop_oneof![
        8 => (0..n, 0..n, 0.0f64..=1.0).prop_map(|(a, b, x)| Op::Observe(a, b, x)),

    /// For any interleaving of observations, probes, audits, ticks and operator
    /// actions: no route ever contains a quarantined link (any mode, cached or not),
    /// AI-mode routes never contain a link above the active reject threshold, and the
    /// safety-state invariants hold after every step.
    #[test]
    fn no_route_ever_uses_a_forbidden_link(ops in proptest::collection::vec(arb_op(), 1..80)) {
        let f = fx();
        let g = complete_graph();
        let cfg = qib_verifier::config::RuntimeConfig::default();
        for op in ops {
            match op {
                Op::Observe(a, b, x) if a != b => { f.dp.observe(NODES[a], NODES[b], x).unwrap(); }
                Op::Probe(a, b, x) if a != b => { let _ = f.dp.ingest_probe(&f.probe(NODES[a], NODES[b], x)); }
                Op::Compare(a, b, x) if a != b => { let _ = f.dp.compare_and_audit(NODES[a], NODES[b], x); }
                Op::Tick(s) => { f.clock.advance_secs(s); f.dp.tick().unwrap(); }
                Op::Conservative(c) => { let _ = f.cp.set_conservative("op", c); }
                Op::Reset => { f.cp.reset_breaker("op").unwrap(); }
                Op::Route(a, b, k) if a != b => {
                    let v = f.dp.safety_view().unwrap();
                    for _ in 0..2 { // second call exercises the cache
                        if let Ok(r) = f.dp.route(NODES[a], NODES[b], k, &g) {
                            prop_assert!(!r.paths.is_empty() && r.paths.len() <= k);
                            for p in &r.paths {
                                prop_assert_eq!(p.nodes.first().unwrap(), NODES[a]);
                                prop_assert_eq!(p.nodes.last().unwrap(), NODES[b]);
                                prop_assert!(p.cost.is_finite() && p.cost >= 0.0);
                                for w in p.nodes.windows(2) {
                                    prop_assert!(!f.dp.is_quarantined(&w[0], &w[1]).unwrap(),
                                        "quarantined hop {}->{} routed in {:?}", w[0], w[1], r.mode);
                                    if r.mode == qib_verifier::cache::RouteMode::Ai {
                                        let lv = f.dp.link(&w[0], &w[1]).unwrap().unwrap();
                                        prop_assert!(lv.std <= cfg.path_reject_threshold + 1e-12,
                                            "AI routed std {} on {}->{}", lv.std, w[0], w[1]);
                                    }
                                    if r.mode == qib_verifier::cache::RouteMode::AiConservative {
                                        let lv = f.dp.link(&w[0], &w[1]).unwrap().unwrap();
                                        prop_assert!(lv.std <= cfg.conservative_reject_threshold + 1e-12);
                                    }
                                }
                            }
                            prop_assert_eq!(f.dp.safety_view().unwrap().version, v.version, "route must not change safety state");
                        }
                    }
                }
                _ => {}
            }
            let v = f.dp.safety_view().unwrap();
            prop_assert_eq!(v.stage == 4, !v.ai_enabled);
            if v.stage >= 3 { prop_assert!(v.conservative); }
            let h = f.dp.health().unwrap();
            prop_assert!((0.0..=1.0).contains(&h.ai_trust_score));
            prop_assert!(h.mean_uncertainty.is_finite());
        }
    }
